const SHEET_PARTES = "Partes"; const SHEET_TRABAJADORES = "Trabajadores"; const SHEET_TRABAJOS = "Trabajos"; const SHEET_DETALLES = "Detalle"; const SHEET_PRODUCCIONES = "Producciones"; // ── doGet: devuelve trabajadores, trabajos y detalles ────────────────── function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const wSheet = ss.getSheetByName(SHEET_TRABAJADORES); const wAll = wSheet.getDataRange().getValues(); const horasAnuales = parseFloat(wAll[0][2]) || 1800; // C1 const workers = wAll.slice(2).filter(r => r[0]).map(r => ({ name: String(r[0]).trim(), tipo: String(r[1] || "Operario").trim(), salario: parseFloat(r[2]) || 0 })); const jSheet = ss.getSheetByName(SHEET_TRABAJOS); const jData = jSheet.getDataRange().getValues().slice(1).filter(r => r[0]); const dSheet = ss.getSheetByName(SHEET_DETALLES); const dData = dSheet.getDataRange().getValues().slice(1).filter(r => r[0] && r[1]); const jobs = {}; jData.forEach(r => { const jobName = String(r[0]).trim(); const label = String(r[1] || "Detalle").trim(); jobs[jobName] = { label, options: [] }; }); dData.forEach(r => { const jobName = String(r[0]).trim(); const opt = String(r[1]).trim(); const pesoBala = parseFloat(r[2]) || 0; if (jobs[jobName]) jobs[jobName].options.push({ name: opt, pesoBala }); }); return ContentService .createTextOutput(JSON.stringify({ ok: true, workers, jobs, horasAnuales })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // ── doPost: escribe en Partes y opcionalmente en Producciones ────────── function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); // Leer horas anuales y salarios const wSheet = ss.getSheetByName(SHEET_TRABAJADORES); const wAll = wSheet.getDataRange().getValues(); const horasAnuales = parseFloat(wAll[0][2]) || 1800; const minAnuales = horasAnuales * 60; const salarioMap = {}; wAll.slice(2).filter(r => r[0]).forEach(r => { salarioMap[String(r[0]).trim()] = parseFloat(r[2]) || 0; }); // ── Hoja Partes ────────────────────────────────────────────────── const partes = ss.getSheetByName(SHEET_PARTES); if (partes.getLastRow() === 0) { partes.appendRow([ "Timestamp","Fecha","Turno","Encargado","Trabajador", "Hora Inicio","Hora Fin","Duracion","Trabajo","Detalle", "Num Balas","Toneladas producidas","Coste por tonelada","Observaciones" ]); partes.getRange(1,1,1,14).setFontWeight("bold"); } // ── Hoja Producciones ──────────────────────────────────────────── const prodSheet = ss.getSheetByName(SHEET_PRODUCCIONES); if (prodSheet.getLastRow() === 0) { prodSheet.appendRow([ "Timestamp","Fecha","Turno","Encargado","Trabajo","Material", "Num Balas","Toneladas Producidas","Num Operarios", "Operarios","Coste Produccion","Observaciones" ]); prodSheet.getRange(1,1,1,12).setFontWeight("bold"); } const BALA_JOBS = ["Embalar 1900","Embalar 2000"]; // Agrupar registros por clave única (mismo trabajo + detalle + t1 + t2) // para consolidar en una sola fila de Producciones por grupo const grupos = {}; // key -> {records, costeTotal, ...} data.records.forEach(r => { const duracion = parseInt(r.duracion_min) || 0; const salario = salarioMap[r.trabajador] || 0; const costePorMin = minAnuales > 0 ? salario / minAnuales : 0; const costeOp = costePorMin * duracion; const numBalas = parseInt(r.num_balas) || 0; const pesoBala = parseFloat(r.peso_bala) || 0; const toneladas = numBalas * pesoBala; const costePorTon = toneladas > 0 ? costeOp / toneladas : 0; // Escribir fila individual en Partes partes.appendRow([ r.timestamp, r.fecha, r.turno, r.encargado, r.trabajador, r.hora_inicio, r.hora_fin, duracion, r.trabajo, r.detalle, numBalas || "", toneladas > 0 ? toneladas.toFixed(1) : "", costePorTon > 0 ? costePorTon.toFixed(2) : "", r.observaciones || "" ]); // Acumular en grupo para Producciones (solo trabajos de embalado) if (BALA_JOBS.includes(r.trabajo) && numBalas > 0) { const key = `${r.fecha}||${r.turno}||${r.trabajo}||${r.detalle}||${r.hora_inicio}||${r.hora_fin}`; if (!grupos[key]) { grupos[key] = { timestamp: r.timestamp, fecha: r.fecha, turno: r.turno, encargado: r.encargado, trabajo: r.trabajo, detalle: r.detalle, numBalas: numBalas, toneladas: toneladas, operarios: [], costeTotal: 0, obs: r.observaciones || "" }; } grupos[key].operarios.push(r.trabajador); grupos[key].costeTotal += costeOp; } }); // Escribir una fila por grupo en Producciones Object.values(grupos).forEach(g => { const numOp = g.operarios.length; const costePorTonGrupo = g.toneladas > 0 ? g.costeTotal / g.toneladas : 0; prodSheet.appendRow([ g.timestamp, g.fecha, g.turno, g.encargado, g.trabajo, g.detalle, g.numBalas, g.toneladas.toFixed(1), numOp, g.operarios.join(" ; "), g.costeTotal.toFixed(2), g.obs ]); }); return ContentService .createTextOutput(JSON.stringify({ ok: true, rows: data.records.length })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
